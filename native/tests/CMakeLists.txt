cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	project(mc_native_l2_tests LANGUAGES C CXX)
	if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
		message(FATAL_ERROR "MineMod Native requires an x86 or x64 toolchain.")
	endif()
	set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
	set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
	include(../cmake/NativeCore.cmake)
	enable_testing()
endif()

add_executable(mc_material_system_abi_test material_system_abi_test.cpp)
target_compile_features(mc_material_system_abi_test PRIVATE cxx_std_17)
add_test(NAME material_system_abi COMMAND mc_material_system_abi_test)

add_executable(mc_native_cpu_work_test cpu_work_gate_test.cpp)
add_executable(mc_resource_source_probe resource_source_probe.cpp)
add_executable(mc_native_atlas_pixels_probe atlas_pixels_probe.cpp)
add_library(mc_native_l2_probe SHARED l2_probe.cpp)
add_executable(mc_native_l2_core_test l2_core_test.cpp)
add_executable(mc_native_l2_pixels_test l2_pixels_test.cpp)
add_executable(mc_pixel_texture_data_test pixel_texture_data_test.cpp)
add_executable(mc_pixel_texture_data_bench pixel_texture_data_bench.cpp)
add_executable(mc_l2_snapshot_bench l2_snapshot_bench.cpp)
add_executable(mc_native_l2_draw_test l2_draw_test.cpp)
add_executable(mc_native_l2_worker_test l2_worker_test.cpp)
add_executable(mc_native_world_core_test world_core_test.cpp)
add_executable(mc_native_world_light_bench world_light_bench.cpp)
add_executable(mc_native_world_worker_test world_worker_test.cpp)
add_executable(mc_native_world_decode_test world_decode_test.cpp)
add_executable(mc_native_world_mesh_test world_mesh_test.cpp)

set(cpu_targets
	mc_native_cpu_work_test mc_resource_source_probe mc_native_atlas_pixels_probe
	mc_native_l2_probe mc_native_l2_core_test mc_native_l2_pixels_test
	mc_pixel_texture_data_test mc_pixel_texture_data_bench mc_l2_snapshot_bench
	mc_native_l2_draw_test mc_native_l2_worker_test mc_native_world_core_test
	mc_native_world_light_bench mc_native_world_worker_test mc_native_world_decode_test
	mc_native_world_mesh_test mc_atlas_regions_test
)
if(WIN32)
	add_executable(mc_native_atlas_sprite_probe atlas_sprite_probe.cpp)
	add_executable(mc_atlas_sampler_dx9 atlas_sampler_dx9.cpp)
	target_link_libraries(mc_atlas_sampler_dx9 PRIVATE d3d9)
	add_executable(mc_material_mips_dx9 material_mips_dx9.cpp)
	target_link_libraries(mc_material_mips_dx9 PRIVATE d3d9 d3dcompiler)
	list(APPEND cpu_targets mc_native_atlas_sprite_probe mc_atlas_sampler_dx9 mc_material_mips_dx9)
endif()
foreach(target IN LISTS cpu_targets)
	target_link_libraries(${target} PRIVATE mc_native_core)
	if(MSVC)
		target_compile_options(${target} PRIVATE /W4 /fp:strict /utf-8)
	else()
		target_compile_options(${target} PRIVATE -Wall -Wextra -ffp-contract=off)
	endif()
endforeach()
