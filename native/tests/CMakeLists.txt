cmake_minimum_required(VERSION 3.20)
project(mc_native_l2_tests LANGUAGES C CXX)

add_library(mc_native_world_decode STATIC ../world_decode.cpp
    ../include/sourcesdk-minimal-x86-64-branch/utils/lzma/C/LzmaDec.c)
target_include_directories(mc_native_world_decode PRIVATE
    ../include/rapidjson-master/include ../include/sourcesdk-minimal-x86-64-branch)
target_compile_features(mc_native_world_decode PRIVATE cxx_std_17)

add_library(mc_native_l2_probe SHARED l2_probe.cpp ../l2_core.cpp ../l2_worker.cpp ../l2_fluid.cpp)
add_executable(mc_native_l2_core_test l2_core_test.cpp ../l2_core.cpp)
add_executable(mc_native_l2_draw_test l2_draw_test.cpp ../l2_draw.cpp ../l2_core.cpp)
add_executable(mc_native_l2_worker_test l2_worker_test.cpp ../l2_worker.cpp ../l2_core.cpp ../l2_fluid.cpp)
add_executable(mc_native_world_core_test world_core_test.cpp ../world_core.cpp ../l2_core.cpp)
add_executable(mc_native_world_worker_test world_worker_test.cpp ../world_session.cpp ../world_worker.cpp ../world_core.cpp ../l2_core.cpp)
add_executable(mc_native_world_decode_test world_decode_test.cpp ../world_session.cpp ../world_worker.cpp ../world_core.cpp ../l2_core.cpp)
add_executable(mc_native_world_mesh_test world_mesh_test.cpp ../world_mesh.cpp ../world_core.cpp ../l2_core.cpp ../l2_worker.cpp ../l2_fluid.cpp)
find_package(Threads REQUIRED)
target_link_libraries(mc_native_l2_draw_test PRIVATE Threads::Threads)
target_link_libraries(mc_native_l2_worker_test PRIVATE Threads::Threads)
target_link_libraries(mc_native_l2_probe PRIVATE Threads::Threads)
target_link_libraries(mc_native_world_worker_test PRIVATE Threads::Threads)
target_link_libraries(mc_native_world_worker_test PRIVATE mc_native_world_decode)
target_link_libraries(mc_native_world_decode_test PRIVATE mc_native_world_decode Threads::Threads)
target_link_libraries(mc_native_world_mesh_test PRIVATE Threads::Threads)
foreach(target mc_native_l2_probe mc_native_l2_core_test mc_native_l2_draw_test mc_native_l2_worker_test mc_native_world_core_test mc_native_world_worker_test mc_native_world_decode_test mc_native_world_mesh_test)
    target_sources(${target} PRIVATE ../l2_shapes.cpp)
    target_compile_features(${target} PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /fp:strict /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -ffp-contract=off)
    endif()
endforeach()
enable_testing()
add_test(NAME l2_core COMMAND mc_native_l2_core_test)
add_test(NAME l2_draw COMMAND mc_native_l2_draw_test)
set_tests_properties(l2_draw PROPERTIES TIMEOUT 30)
add_test(NAME l2_workers COMMAND mc_native_l2_worker_test)
add_test(NAME world_core COMMAND mc_native_world_core_test)
add_test(NAME world_workers COMMAND mc_native_world_worker_test)
add_test(NAME world_decode COMMAND mc_native_world_decode_test)
add_test(NAME world_mesh COMMAND mc_native_world_mesh_test)
set_tests_properties(world_mesh PROPERTIES TIMEOUT 30)
set_tests_properties(world_decode PROPERTIES TIMEOUT 30)
set_tests_properties(world_workers PROPERTIES TIMEOUT 30)
set_tests_properties(world_core PROPERTIES TIMEOUT 60)
set_tests_properties(l2_workers PROPERTIES TIMEOUT 30)
